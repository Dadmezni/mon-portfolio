function cosineSimilarity(array $a, array $b): float
{
$dot = $normA = $normB = 0.0;
foreach ($a as $i => $x) {
$dot += $x * $b[$i];
$normA += $x * $x;
$normB += $b[$i] * $b[$i];
}
return ($normA && $normB) ? $dot / (sqrt($normA) * sqrt($normB)) : 0.0;
}
